    nicknameTopMargin = 100,       -- Pixels from top of screen
    messageMargin = 40,            -- Pixels below nickname for message

-- Effect helpers (applied on respective realms)
local EFFECTS = {
    HEAL = function(ply, amount)
        if SERVER and isValid(ply) then
            local newHealth = math.min(ply:getHealth() + amount, ply:getMaxHealth())
            ply:setHealth(newHealth)
        end
    end,
    
    ARMOR = function(ply, amount)
        if SERVER and isValid(ply) then
            local newArmor = math.min(ply:getArmor() + amount, ply:getMaxArmor())
            ply:setArmor(newArmor)
        end
    end,    
    
    MATERIAL = function(ply, matName, duration)
        if CLIENT and isValid(ply) then
            local originalMat = ply:getMaterial()
            ply:setMaterial(matName)
            
            timer.simple(duration, function()
                if isValid(ply) then
                    ply:setMaterial(originalMat)
                end
            end)
        end
    end,
    
    GODMODE = function(ply, duration)
        -- Client: visual effect
        if CLIENT and isValid(ply) then
            local originalMat = ply:getMaterial()
            local originalColor = ply:getColor()
            
            ply:setMaterial("models/debug/debugwhite")
            ply:setColor(Color(255, 215, 0))
            
            timer.simple(duration, function()
                if isValid(ply) then
                    ply:setMaterial(originalMat)
                    ply:setColor(originalColor)
                end
            end)
        end
        
        -- Server: health boost
        if SERVER and isValid(ply) then
            ply:setHealth(10000)
            timer.simple(duration, function()
                if isValid(ply) then
                    ply:setHealth(100)
                end
            end)
        end
    end
}

-- ============= SERVER LOGIC =============
if SERVER then
    local killStreaks = {}
    local lastKillInfo = {}
    
    -- Apply server-side effects
    local function applyEffect(effect, ply)
        if effect.type == "HEAL" and EFFECTS.HEAL then
            EFFECTS.HEAL(ply, unpack(effect.args or {}))
        elseif effect.type == "GODMODE" and EFFECTS.GODMODE then
            EFFECTS.GODMODE(ply, unpack(effect.args or {}))
        end
    end
    
    hook.add("PlayerDeath", "KillstreakTracker", function(victim, inflictor, attacker)
        if not isValid(victim) then return end
        
        local isSuicide = not isValid(attacker) or attacker == victim
        local weaponClass = isValid(inflictor) and inflictor:getClass() or "unknown"
        
        lastKillInfo[victim] = {
            time = timer.curtime(),
            weapon = weaponClass,
            isSuicide = isSuicide
        }
        
        killStreaks[victim] = 0
        
        if isSuicide then
            return
        end
        
        if isValid(attacker) and attacker:isPlayer() then
            killStreaks[attacker] = (killStreaks[attacker] or 0) + 1
            local streak = killStreaks[attacker]
            
            lastKillInfo[attacker] = {
                time = timer.curtime(),
                weapon = weaponClass,
                victim = victim
            }
            
            local triggeredLevels = {}
            
            for _, level in ipairs(LEVELS) do
                local trigger = level.trigger
                local matched = false
                
                if trigger.type == "streak" and streak == trigger.count then
                    matched = true
                elseif trigger.type == "weapon" and weaponClass == trigger.class then
                    matched = true
                end
                
                if matched then
                    table.insert(triggeredLevels, level)
                end
            end
            
            for _, level in ipairs(triggeredLevels) do
                net.start("KillstreakNotify")
                net.writeEntity(attacker)
                net.writeTable(level)
                net.send()
                
                -- Apply server effects
                if level.effects then
                    for _, effect in ipairs(level.effects) do
                        local target = effect.target == "victim" and victim or attacker
                        applyEffect(effect, target)
                    end
                end
            end
        end
    end)
    
    -- Player toggle preferences
    local playerToggles = {}
    
    hook.add("PlayerSay", "KillstreakCommands", function(ply, text)
        if text == "!kstoggle" or text == "!killstreak" then
            playerToggles[ply] = not playerToggles[ply]
            local state = playerToggles[ply] and "disabled" or "enabled"
            printHud(ply, Color(255, 150, 0), "[Killstreak] ", Color(255, 255, 255), "Notifications " .. state)
            
            net.start("KillstreakToggle")
            net.writeBool(playerToggles[ply])
            net.send(ply)
            return ""
        end
        
        if ply ~= owner() then return end
        
        if text == "!ksdebug" then
            local randomLevel = LEVELS[math.random(1, #LEVELS)]
            
            net.start("KillstreakNotify")
            net.writeEntity(owner())
            net.writeTable(randomLevel)
            net.send()
            
            printHud(owner(), Color(0, 255, 0), "[Killstreak] ", Color(255, 255, 255), "Triggered: " .. randomLevel.id)
            return ""
        end
    end)
end

-- ============= CLIENT LOGIC =============
if CLIENT then
    -- Notification queue system
    local activeNotification = nil
    local notificationQueue = {}
    local notificationFont = render.createFont("Arial", 48, 1000, true)
    local messageFont = render.createFont("Arial", 36, 800, true)
    
    -- Cached killstreak images
    local killstreakImages = {}
    
    -- Particle system
    local totalParticleCount = 0
    
    -- Player toggle (disabled = true means NO notifications)
    local isDisabled = false
    
    -- HSV to RGB conversion
    local function hsvToRGB(h, s, v)
        local r, g, b
        
        local c = v * s
        local x = c * (1 - math.abs((h / 60) % 2 - 1))
        local m = v - c
        
        if h < 60 then
            r, g, b = c, x, 0
        elseif h < 120 then
            r, g, b = x, c, 0
        elseif h < 180 then
            r, g, b = 0, c, x
        elseif h < 240 then
            r, g, b = 0, x, c
        elseif h < 300 then
            r, g, b = x, 0, c
        else
            r, g, b = c, 0, x
        end
        
        return (r + m) * 255, (g + m) * 255, (b + m) * 255
    end
    
    -- Create particle burst
    local function createParticleBurst(ply, effectType)
        if not isValid(ply) then return end
        if totalParticleCount >= CONFIG.particleLimit then return end
        
        local pos = ply:getPos() + Vector(0, 0, 40)
        local emitter = particle.create(pos, false)
        if not emitter then return end
        
        local particleCount = 0
        local maxParticles = math.min(50, CONFIG.particleLimit - totalParticleCount)
        local mat = material.load("effects/spark")
        
        if effectType == "PARTICLES_EXPLOSION_RED" then
            -- Red slow explosion
            local lifetime = 1.5
            for i = 1, maxParticles do
                if totalParticleCount >= CONFIG.particleLimit then break end
                
                local angle = math.rad(math.random(0, 360))
                local pitch = math.rad(math.random(-45, 45))
                local speed = math.random(50, 150)
                
                local vel = Vector(
                    math.cos(angle) * math.cos(pitch) * speed,
                    math.sin(angle) * math.cos(pitch) * speed,
                    math.sin(pitch) * speed
                )
                
                local size = math.random(2, 5)
                local p = emitter:add(mat, pos, size, size, 0, 0, 255, 0, lifetime)
                if p then
                    p:setVelocity(vel)
                    p:setGravity(Vector(0, 0, -200))
                    p:setColor(Color(255, math.random(0, 50), 0))
                    particleCount = particleCount + 1
                    totalParticleCount = totalParticleCount + 1
                end
            end
            
        elseif effectType == "PARTICLES_EXPLOSION_RGB" then
            -- Rainbow fast explosion
            local lifetime = 1.0
            for i = 1, maxParticles do
                if totalParticleCount >= CONFIG.particleLimit then break end
                
                local angle = math.rad(math.random(0, 360))
                local pitch = math.rad(math.random(-45, 45))
                local speed = math.random(150, 300)
                
                local vel = Vector(
                    math.cos(angle) * math.cos(pitch) * speed,
                    math.sin(angle) * math.cos(pitch) * speed,
                    math.sin(pitch) * speed
                )
                
                local hue = math.random(0, 360)
                local r, g, b = hsvToRGB(hue, 1, 1)
                
                local size = math.random(3, 7)
                local p = emitter:add(mat, pos, size, size, 0, 0, 255, 0, lifetime)
                if p then
                    p:setVelocity(vel)
                    p:setGravity(Vector(0, 0, -300))
                    p:setColor(Color(r, g, b))
                    particleCount = particleCount + 1
                    totalParticleCount = totalParticleCount + 1
                end
            end
            
        elseif effectType == "PARTICLES_DRIP" then
            -- Drip down from above
            local lifetime = 2.0
            local topPos = pos + Vector(0, 0, 200)
            for i = 1, maxParticles do
                if totalParticleCount >= CONFIG.particleLimit then break end
                
                local offset = Vector(math.random(-50, 50), math.random(-50, 50), 0)
                local spawnPos = topPos + offset
                local hue = math.random(0, 360)
                local r, g, b = hsvToRGB(hue, 1, 1)
                
                local size = math.random(2, 4)
                local p = emitter:add(mat, spawnPos, size, size, 0, 0, 255, 0, lifetime)
                if p then
                    p:setVelocity(Vector(0, 0, math.random(-100, -200)))
                    p:setGravity(Vector(0, 0, -100))
                    p:setColor(Color(r, g, b))
                    particleCount = particleCount + 1
                    totalParticleCount = totalParticleCount + 1
                end
            end
            
        elseif effectType == "PARTICLES_DRIP_UP" then
            -- Drip up from below
            local lifetime = 2.0
            local bottomPos = pos - Vector(0, 0, 100)
            for i = 1, maxParticles do
                if totalParticleCount >= CONFIG.particleLimit then break end
                
                local offset = Vector(math.random(-50, 50), math.random(-50, 50), 0)
                local spawnPos = bottomPos + offset
                local hue = math.random(0, 360)
                local r, g, b = hsvToRGB(hue, 1, 1)
                
                local size = math.random(2, 4)
                local p = emitter:add(mat, spawnPos, size, size, 0, 0, 255, 0, lifetime)
                if p then
                    p:setVelocity(Vector(0, 0, math.random(100, 200)))
                    p:setGravity(Vector(0, 0, 100))
                    p:setColor(Color(r, g, b))
                    particleCount = particleCount + 1
                    totalParticleCount = totalParticleCount + 1
                end
            end
        end
        
        -- Cleanup emitter after particles die
        local maxLifetime = effectType == "PARTICLES_EXPLOSION_RGB" and 2 or 3
        timer.simple(maxLifetime, function()
            if emitter and emitter.destroy then
                emitter:destroy()
            end
            totalParticleCount = math.max(0, totalParticleCount - particleCount)
        end)
    end
    
    -- Apply client-side effects
    local function applyEffect(effect, ply)
        if effect.type == "MATERIAL" and EFFECTS.MATERIAL then
            EFFECTS.MATERIAL(ply, unpack(effect.args or {}))
        elseif effect.type == "GODMODE" and EFFECTS.GODMODE then
            EFFECTS.GODMODE(ply, unpack(effect.args or {}))
        elseif effect.type == "PARTICLES_EXPLOSION_RED" then
            createParticleBurst(ply, "PARTICLES_EXPLOSION_RED")
        elseif effect.type == "PARTICLES_EXPLOSION_RGB" then
            createParticleBurst(ply, "PARTICLES_EXPLOSION_RGB")
        elseif effect.type == "PARTICLES_DRIP" then
            createParticleBurst(ply, "PARTICLES_DRIP")
        elseif effect.type == "PARTICLES_DRIP_UP" then
            createParticleBurst(ply, "PARTICLES_DRIP_UP")
        end
    end
    
    -- Load killstreak image
    local function loadKillstreakImage(levelId, soundFile)
        if killstreakImages[levelId] then return end
        
        -- Convert sound filename to image filename
        local imageName = soundFile:gsub(".wav$", ".png")
        local imageURL = CONFIG.githubImagesBase .. imageName
        
        -- Load material from URL
        render.createMaterial(imageURL, function(mat, url, w, h)
            if mat then
                killstreakImages[levelId] = mat
            end
        end)
    end
    
    -- Play next notification from queue
    local function playNextNotification()
        if activeNotification then return end
        if #notificationQueue == 0 then return end
        if isDisabled then return end -- Skip if disabled
        
        -- Take first from queue
        local notif = table.remove(notificationQueue, 1)
        notif.startTime = timer.curtime()
        activeNotification = notif
        
        -- Load image
        loadKillstreakImage(notif.level.id, notif.level.sound)
        
        -- Play sound (only if not disabled)
        if not isDisabled then
            local soundURL = CONFIG.githubSoundsBase .. notif.level.sound
            local soundOrigin = notif.level.play_at == "victim" and notif.player or notif.player
            
            local is2DSound = notif.level.trigger and notif.level.trigger.count and notif.level.trigger.count > CONFIG.sound2DAfterStreak
            local soundFlags = is2DSound and "noblock" or "3d noblock"
            
            if isValid(soundOrigin) then
                bass.loadURL(soundURL, soundFlags, function(snd, err, errName)
                    if snd then
                        snd:setVolume(CONFIG.soundVolume)
                        if not is2DSound then
                            snd:setPos(soundOrigin:getPos())
                        end
                        snd:play()
                    end
                end)
            end
        end
        
        -- Apply effects
        if notif.level.effects then
            for _, effect in ipairs(notif.level.effects) do
                local target = effect.target == "victim" and notif.player or notif.player
                applyEffect(effect, target)
            end
        end
    end
    
    -- Queue management
    hook.add("think", "KillstreakQueue", function()
        if not activeNotification then
            playNextNotification()
        else
            -- Check if expired
            local age = timer.curtime() - activeNotification.startTime
            if age > CONFIG.notificationDuration then
                activeNotification = nil
                playNextNotification()
            end
        end
    end)
    
    -- Get nickname color based on effect
    local function getNicknameColor(effect, baseColor, time)
        local col = Color(baseColor.r, baseColor.g, baseColor.b)
        
        if effect == "none" then
            return col
        elseif effect == "pulse" then
            local pulse = math.abs(math.sin(time * 3))
            return Color(
                col.r * (0.5 + pulse * 0.5),
                col.g * (0.5 + pulse * 0.5),
                col.b * (0.5 + pulse * 0.5)
            )
        elseif effect == "rgb" then
            local hue = (time * 60) % 360
            local r, g, b = hsvToRGB(hue, 1, 1)
            return Color(r, g, b)
        elseif effect == "rgb_fast" then
            local hue = (time * 120) % 360
            local r, g, b = hsvToRGB(hue, 1, 1)
            return Color(r, g, b)
        elseif effect == "shake" then
            return col
        elseif effect == "shake_rgb" then
            local hue = (time * 90) % 360
            local r, g, b = hsvToRGB(hue, 1, 1)
            return Color(r, g, b)
        end
        return col
    end
    
    -- Get shake offset
    local function getShakeOffset(effect, time)
        if effect == "shake" or effect == "shake_rgb" then
            return Vector(
                math.sin(time * 30) * 5,
                math.cos(time * 25) * 5,
                0
            )
        end
        return Vector(0, 0, 0)
    end
    
    -- Network receivers
    net.receive("KillstreakNotify", function()
        local ply = net.readEntity()
        local level = net.readTable()
        
        if not isValid(ply) then return end
        if CONFIG.debugMode and player() ~= owner() then return end
        if isDisabled then return end -- Skip if disabled
        
        -- Add to queue
        table.insert(notificationQueue, {
            player = ply,
            level = level,
            startTime = timer.curtime(),
            nickname = ply:getName()
        })
        
        -- Try to play immediately
        playNextNotification()
    end)
    
    net.receive("KillstreakToggle", function()
        isDisabled = net.readBool()
        if isDisabled then
            -- Clear queue and active notification
            notificationQueue = {}
            activeNotification = nil
        end
    end)
    
    -- Draw HUD
    hook.add("drawhud", "KillstreakNotifications", function()
        if not activeNotification then return end
        if isDisabled then return end -- Don't draw if disabled
        
        local now = timer.curtime()
        local scrW, scrH = render.getGameResolution()
        
        local notif = activeNotification
        local age = now - notif.startTime
        local level = notif.level
        
        -- Calculate fade alpha
        local alpha = 255
        if age > CONFIG.notificationDuration - 1 then
            alpha = math.floor(255 * (1 - (age - (CONFIG.notificationDuration - 1))))
        end
        
        local yPos = CONFIG.nicknameTopMargin
        local shake = getShakeOffset(level.nickname_effect, now)
        
        -- Draw nickname
        render.setFont(notificationFont)
        local nicknameColor = getNicknameColor(level.nickname_effect, level.nickname_color, now)
        nicknameColor.a = alpha
        render.setColor(nicknameColor)
        render.drawSimpleText(
            scrW / 2 + shake.x,
            yPos + shake.y,
            notif.nickname,
            1, 1
        )
        
        -- Draw message
        render.setFont(messageFont)
        render.setColor(Color(255, 255, 255, alpha))
        local messageY = yPos + CONFIG.messageMargin
        render.drawSimpleText(
            scrW / 2 + shake.x,
            messageY + shake.y,
            level.message,
            1, 1
        )
        
        -- Draw killstreak image BELOW message
        local imgY = messageY + CONFIG.imageMargin
        local mat = killstreakImages[level.id]
        if mat then
            local imgW = CONFIG.imageWidth
            local imgH = CONFIG.imageHeight
            local imgX = scrW / 2 - imgW / 2
            
            render.setColor(Color(255, 255, 255, alpha))
            render.setMaterial(mat)
            render.drawTexturedRect(imgX, imgY, imgW, imgH)
        else
            -- Fallback: colored placeholder while loading
            render.setColor(Color(level.nickname_color.r, level.nickname_color.g, level.nickname_color.b, alpha * 0.3))
            render.drawRect(scrW / 2 - 100, imgY, 200, 150)
            
            -- Loading text
            render.setFont(messageFont)
            render.setColor(Color(255, 255, 255, alpha * 0.5))
            render.drawSimpleText(scrW / 2, imgY + 75, "Loading...", 1, 1)
        end
    end)
    
    enableHud(player(), true)
end
