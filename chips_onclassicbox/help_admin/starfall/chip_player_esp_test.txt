            net.writeBool(impactDebugEnabled)
            net.writeBool(chamsEnabled)
            net.send(owner())
            
            return ""
        end

    -- Get chip quota on server
    local function getChipQuota(ent)
        if not isValid(ent) then return 0 end
        local quota = 0
        pcall(function()
            quota = ent:getQuotaAverage() or 0
        end)
        return math.floor(quota * 1000000)
    end
    
    -- Collect and send chip data to client
    timer.create("ESP_SendChipData", 1, 0, function()
        if not chipsESPEnabled then return end
        
        local chipData = {}
        local ownerPos = owner():getPos()
        
        -- Collect SF chips
        for _, sf in pairs(find.byClass("starfall_processor")) do
            if isValid(sf) then
                local quota = getChipQuota(sf)
                local distance = sf:getPos():getDistance(ownerPos)
                
                -- Skip low-usage distant chips (less than 300us and further than 3000 units)
                if quota < 300 and distance > 3000 then continue end
                
                local chipOwner = sf:getOwner()
                table.insert(chipData, {
                    entIndex = sf:entIndex(),
                    chipType = "SF",
                    name = sf:getChipName() ~= "" and sf:getChipName() or "<no name>",
                    ownerName = isValid(chipOwner) and chipOwner:getName() or "disconnected",
                    ownerTeam = isValid(chipOwner) and chipOwner:getTeam() or 0,
                    quota = quota
                })
            end
        end
        
        -- Collect E2 chips
        for _, e2 in pairs(find.byClass("gmod_wire_expression2")) do
            if isValid(e2) then
                local quota = getChipQuota(e2)
                local distance = e2:getPos():getDistance(ownerPos)
                
                -- Skip low-usage distant chips (less than 300us and further than 3000 units)
                if quota < 300 and distance > 3000 then continue end
                
                local chipOwner = e2:getOwner()
                table.insert(chipData, {
                    entIndex = e2:entIndex(),
                    chipType = "E2",
                    name = e2:getChipName() ~= "" and e2:getChipName() or "<no name>",
                    ownerName = isValid(chipOwner) and chipOwner:getName() or "disconnected",
                    ownerTeam = isValid(chipOwner) and chipOwner:getTeam() or 0,
                    quota = quota
                })
            end
        end
        
        -- Collect white props
        local whitePropData = {}
        for _, ent in pairs(find.byClass("prop_physics")) do
            if isValid(ent) then
                local col = ent:getColor()
                
                if col.r == 255 and col.g == 255 and col.b == 255 and col.a == 100 then
                    local entOwner = ent:getOwner()
                    table.insert(whitePropData, {
                        entIndex = ent:entIndex(),
                        ownerName = isValid(entOwner) and entOwner:getName() or "disconnected",
                        ownerTeam = isValid(entOwner) and entOwner:getTeam() or 0
                    })
                end
            end
        end
        
        net.start("ESP_ChipData")
        net.writeTable(chipData)
        net.writeTable(whitePropData)
        net.send(owner())
    end)
    
    -- Collect and send player data to client
    timer.create("ESP_SendPlayerData", 2, 0, function()
        if not playersESPEnabled then return end
        
        local playerData = {}
        
        for _, ply in pairs(find.allPlayers()) do
            if isValid(ply) and ply ~= owner() then
                local mode = "UNKNOWN"
                
                if ply.isPVP and ply:isPVP() then
                    mode = "PVP"
                elseif ply.isBUILD and ply:isBUILD() then
                    mode = "BUILD"
                end
                
                -- Count entities by owner (more reliable on server)
                local entCount = 0
                for _, ent in pairs(find.all()) do
                    if isValid(ent) then
                        local entOwner = ent:getOwner()
                        if isValid(entOwner) and entOwner == ply then
                            entCount = entCount + 1
                        end
                    end
                end
                
                -- Get hours
                local secondsPlayed = 0
                if ply.getTimePlayed then
                    secondsPlayed = ply:getTimePlayed() or 0
                end
                local hoursPlayed = math.floor(secondsPlayed / 3600)
                
                table.insert(playerData, {
                    entIndex = ply:entIndex(),
                    name = ply:getName(),
                    team = ply:getTeam(),
                    mode = mode,
                    entCount = entCount,
                    hoursPlayed = hoursPlayed
                })
            end
        end
        
        net.start("ESP_PlayerData")
        net.writeTable(playerData)
        net.send(owner())
    end)
end

    -- Get color based on CPU usage (green to red gradient)
    local function getCPUColor(usage)
        if usage <= 500 then return Color(0, 255, 0) end -- Green for low usage
        if usage >= 2000 then return Color(255, 0, 0) end -- Red for high usage
        
        local ratio = (usage - 500) / 1500
        local r = math.floor(255 * ratio)
        local g = math.floor(255 * (1 - ratio))
        return Color(r, g, 0)
    end
    
    -- Draw corner brackets around text
    local function drawCornerBrackets(x, y, w, h, color, thickness)
        local cornerSize = 8
        thickness = thickness or 2
        
        render.setColor(color)
        
        -- Top-left corner
        render.drawRectFast(x, y, cornerSize, thickness)
        render.drawRectFast(x, y, thickness, cornerSize)
        
        -- Top-right corner
        render.drawRectFast(x + w - cornerSize, y, cornerSize, thickness)
        render.drawRectFast(x + w - thickness, y, thickness, cornerSize)
        
        -- Bottom-left corner
        render.drawRectFast(x, y + h - thickness, cornerSize, thickness)
        render.drawRectFast(x, y + h - cornerSize, thickness, cornerSize)
        
        -- Bottom-right corner
        render.drawRectFast(x + w - cornerSize, y + h - thickness, cornerSize, thickness)
        render.drawRectFast(x + w - thickness, y + h - cornerSize, thickness, cornerSize)
    end
    
    -- Get gradient color for entity count (green to red)
    local function getEntityCountColor(count)
        if count <= 0 then return Color(0, 255, 0) end
        if count >= 200 then return Color(255, 0, 0) end
        
        local ratio = count / 200
        local r = math.floor(255 * ratio)
        local g = math.floor(255 * (1 - ratio))
        return Color(r, g, 0)
    end
    
    -- Get gradient color for hours (red -> yellow -> green)
    local function getHoursColor(hours)
        if hours <= 0 then return Color(255, 0, 0) end
        if hours >= 300 then return Color(0, 255, 0) end
        
        if hours <= 100 then
            local ratio = hours / 100
            return Color(255, math.floor(255 * ratio), 0)
        else
            local ratio = (hours - 100) / 200
            return Color(math.floor(255 * (1 - ratio)), 255, 0)
        end
    end

    -- ============= CHAMS SYSTEM =============
    -- Generate random color for player
    local function getPlayerChamsColor(ply)
        if not playerChamsColors[ply] then
            -- Generate bright random color
            playerChamsColors[ply] = Color(
                math.random(100, 255),
                math.random(100, 255),
                math.random(100, 255)
            )
        end
        return playerChamsColors[ply]
    end
    
    -- Store original materials
    local originalMaterials = {}
    local originalColors = {}
    
    -- Create flat material
    local chamsMat = material.create("UnlitGeneric")
    chamsMat:setInt("$flags", 256) -- Ignore Z
    
    -- Apply/remove chams materials
    hook.add("think", "ChamsApply", function()
        if not chamsEnabled then
            -- Restore originals
            for ply, mat in pairs(originalMaterials) do
                if isValid(ply) then
                    ply:setMaterial(mat)
                    if originalColors[ply] then
                        ply:setColor(originalColors[ply])
                    end
                end
            end
            originalMaterials = {}
            originalColors = {}
            return
        end
        if cpuTotalAverage() > cpuMax() * 0.9 then return end
        
        for _, ply in ipairs(find.allPlayers()) do
            if not isValid(ply) then continue end
            if ply == owner() then continue end
            
            -- Store original once
            if not originalMaterials[ply] then
                originalMaterials[ply] = ply:getMaterial()
                originalColors[ply] = ply:getColor()
            end
            
            local chamsColor = getPlayerChamsColor(ply)
            
            -- Apply color and flat material (will render through walls due to $flags)
            ply:setColor(chamsColor)
            ply:setMaterial("!chams_" .. ply:entIndex())
        end
    end)
    
    -- Custom render for chams with proper depth
    hook.add("postdrawentities", "ChamsRenderCustom", function()
        if not chamsEnabled then return end
        if cpuTotalAverage() > cpuMax() * 0.9 then return end

        render.setMaterial(chamsMat)
        
        for _, ply in ipairs(find.allPlayers()) do
            if not isValid(ply) then continue end
            if ply == owner() then continue end
            
            local chamsColor = getPlayerChamsColor(ply)
            render.setColorModulation(chamsColor.r / 255, chamsColor.g / 255, chamsColor.b / 255)
        end
        
        render.setColorModulation(1, 1, 1)
    end)

    -- ============= IMPACT DEBUG SYSTEM =============
    local impactMarkers = {} -- {pos, time, isHit, startPos}
    local nearbyEntities = {} -- Cache nearby entities for hitbox rendering
    local lastWeaponFire = 0
    local lastHitboxUpdate = 0
    local HITBOX_RADIUS = 500
    local IMPACT_LIFETIME = 5
    
    -- Update nearby entities for hitbox rendering
    local function updateNearbyEntities()
        if not impactDebugEnabled then
            nearbyEntities = {}
            return
        end
        if not isValid(owner()) then return end
        
        local ownerPos = owner():getPos()
        nearbyEntities = find.inSphere(ownerPos, HITBOX_RADIUS)
        
        -- Always include all players regardless of distance for eye trace
        for _, ply in ipairs(find.allPlayers()) do
            if isValid(ply) and ply ~= owner() then
                if not table.hasValue(nearbyEntities, ply) then
                    table.insert(nearbyEntities, ply)
                end
            end
        end
    end
    
    -- Hook weapon firing using InputPressed
    hook.add("InputPressed", "ImpactDebug_Input", function(button)
        if not impactDebugEnabled then return end
        
        -- Check if it's left mouse button (MOUSE.LEFT = 107)
        if button ~= MOUSE.LEFT and button ~= MOUSE.RIGHT then return end
        
        local now = timer.curtime()
        if now - lastWeaponFire < 0.05 then return end -- Debounce
        lastWeaponFire = now
        
        -- Perform trace from eye position
        local eyePos = owner():getEyePos()
        local eyeAng = owner():getEyeAngles()
        local endPos = eyePos + eyeAng:getForward() * 50000
        
        -- Create filter that includes players
        local tr = trace.trace(eyePos, endPos, owner())
        
        if tr.Hit then
            table.insert(impactMarkers, {
                pos = tr.HitPos,
                time = now,
                isHit = isValid(tr.Entity),
                startPos = eyePos
            })
        end
    end)
    
    -- Update nearby entities periodically
    hook.add("think", "ImpactDebug_UpdateEnts", function()
        if not impactDebugEnabled then return end
        if cpuTotalAverage() > cpuMax() * 0.9 then return end
        local now = timer.curtime()
        if now - lastHitboxUpdate > 0.5 then
            lastHitboxUpdate = now
            updateNearbyEntities()
        end
    end)
    
    -- Render hitboxes and impact markers
    hook.add("postdrawopaquerenderables", "ImpactDebug_Draw", function()
        if not impactDebugEnabled then return end
        if cpuTotalAverage() > cpuMax() * 0.9 then return end
        local now = timer.curtime()
        
        -- Draw impact markers and bullet traces
        for i = #impactMarkers, 1, -1 do
            local impact = impactMarkers[i]
            local age = now - impact.time
            
            if age > IMPACT_LIFETIME then
                table.remove(impactMarkers, i)
            else
                local alpha = math.floor(255 * (1 - age / IMPACT_LIFETIME))
                
                -- Draw bullet trace line (green)
                render.setColor(Color(0, 255, 0, alpha))
                render.draw3DLine(impact.startPos, impact.pos, true)
                
                -- Draw impact point as wireframe box
                local impactColor = impact.isHit and Color(255, 0, 0) or Color(0, 100, 255)
                render.setColor(Color(impactColor.r, impactColor.g, impactColor.b, alpha))
                render.draw3DWireframeBox(impact.pos, Angle(0,0,0), Vector(-3,-3,-3), Vector(3,3,3), true)
            end
        end
        
        -- Draw player eye traces
        for _, ply in ipairs(find.allPlayers()) do
            if not isValid(ply) then continue end
            if ply == owner() then continue end
            
            local eyePos = ply:getEyePos()
            local eyeAng = ply:getEyeAngles()
            local endPos = eyePos + eyeAng:getForward() * 10000
            local tr = trace.line(eyePos, endPos, ply)
            
            if tr.Hit then
                -- Check if looking at a player
                local isLookingAtPlayer = isValid(tr.Entity) and tr.Entity:isPlayer()
                local traceColor = isLookingAtPlayer and Color(255, 0, 0, 150) or Color(100, 100, 255, 80)
                
                render.setColor(traceColor)
                render.draw3DLine(eyePos, tr.HitPos, true)
            end
        end
        
        -- Draw OBB hitboxes for nearby entities with color coding
        for _, ent in ipairs(nearbyEntities) do
            if not isValid(ent) then continue end
            if ent == owner() then continue end
            
            local hitboxColor = Color(255, 255, 255, 100)
            
            -- Special handling for players - use hull bounds instead of OBB
            if ent:isPlayer() then
                local teamColor = team.getColor(ent:getTeam())
                hitboxColor = Color(teamColor.r, teamColor.g, teamColor.b, 150)
                
                -- Use hull mins/maxs for accurate player hitbox
                local mins = ent:obbMins()
                local maxs = ent:obbMaxs()
                
                -- Draw player capsule approximation
                local pos = ent:getPos()
                local ang = ent:getAngles()
                
                render.setColor(hitboxColor)
                render.draw3DWireframeBox(pos, ang, mins, maxs, true)
                
            elseif ent:isNPC() then
                hitboxColor = Color(255, 150, 0, 120)
                local mins, maxs = ent:obbMins(), ent:obbMaxs()
                render.setColor(hitboxColor)
                render.draw3DWireframeBox(ent:getPos(), ent:getAngles(), mins, maxs, true)
                
            elseif ent:isVehicle() then
                hitboxColor = Color(0, 255, 255, 120)
                local mins, maxs = ent:obbMins(), ent:obbMaxs()
                render.setColor(hitboxColor)
                render.draw3DWireframeBox(ent:getPos(), ent:getAngles(), mins, maxs, true)
                
            else
                -- Props and other entities
                local entOwner = ent:getOwner()
                if isValid(entOwner) then
                    if entOwner == owner() then
                        hitboxColor = Color(0, 255, 0, 120)
                    else
                        local teamColor = team.getColor(entOwner:getTeam())
                        hitboxColor = Color(teamColor.r, teamColor.g, teamColor.b, 100)
                    end
                else
                    hitboxColor = Color(150, 150, 150, 80)
                end
                
                local mins, maxs = ent:obbMins(), ent:obbMaxs()
                render.setColor(hitboxColor)
                render.draw3DWireframeBox(ent:getPos(), ent:getAngles(), mins, maxs, true)
            end
        end
    end)
    -- ============= END IMPACT DEBUG SYSTEM =============
    
    -- Enable HUD
    enableHud(player(), true)
    
end
