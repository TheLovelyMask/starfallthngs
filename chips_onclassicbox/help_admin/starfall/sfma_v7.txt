SFMA.PhysicsLag = {
    enabled = true,
    checkRate = 0.15,               -- how often to check for lag (seconds)
    freezeThreshold = 0.75,         -- 75% of props frozen = lag detected
    minPropsForDetection = 20,      -- minimum props on map to enable detection

    -- Scoring weights for offender detection
    weights = {
        constraints = 0.05,          -- points per constraint
        velocitySpike = 100,          -- points per velocity spike
        heavyProp = 1.5,            -- points per heavy prop (>500kg)
        recentSpawn = 3,            -- points per recent spawn
        whiteProp = 100,              -- points per white-colored prop (lag indicator)

    heavyPropMass = 500,            -- kg threshold for "heavy" prop
    recentSpawnWindow = 5,          -- seconds to consider "recent"
    velocitySpikeThreshold = 2000,  -- velocity change to count as spike
}

function SFMA.logOffender( plr, ping )
    if ping == nil then ping = false end

    SFMA.lastOffender = plr

    timer.remove("SFMA_clearLastOffender")
    timer.create("SFMA_clearLastOffender", 30, 1, function()
        SFMA.lastOffender = nil
    end)

    if SERVER then
        -- send net to owner only (owner receives and will show HUD)
        pcall(function()
            net.start("SFMA_logOffender")
            net.writeEntity(plr)
            net.writeBool(ping)
            net.send(owner())
        end)
    end

    if SERVER and SFMA.autoSpectate and ping and owner() and isValid(owner()) then
        -- only the owner should trigger server-side autospectate behavior or use it as admin action
        concmd("fspectate \"" .. plr:getName() .. "\"")
    end
end

-- ======================== SERVER LOGIC ========================
if SERVER then

    -- ============= PHYSICS LAG DETECTION SYSTEM =============
    local lastVelocities = {}           -- track velocity changes
    local collisionCounts = {}          -- track collisions per player
    local lastLagReport = 0             -- cooldown for lag reports
    local entitySpawnTimes = {}         -- track when entities were created
    local canaryProp = nil              -- test prop to detect server freezes
    local canaryWasFrozen = false       -- track previous freeze state
    
    -- Forward declare function
    local scanForPhysicsOffender
    
    -- Create canary prop (falling prop that shouldn't freeze unless server lags)
    local function createCanaryProp()
        if not SERVER then return end
        
        -- Remove old canary if exists
        if isValid(canaryProp) then
            canaryProp:remove()
        end
        
        -- Create prop above chip
        local chipPos = chip():getPos()
        local spawnPos = chipPos + Vector(0, 0, 150)
        
        canaryProp = prop.create(spawnPos, Angle(0, 0, 0), "models/hunter/blocks/cube025x025x025.mdl", false)
        
        if isValid(canaryProp) then
            canaryProp:setNoDraw(false)  -- Make it invisible
            canaryProp:setCollisionGroup(COLLISION_GROUP.DEBRIS)  -- Don't collide with players
            
            local phys = canaryProp:getPhysicsObject()
            if phys and phys:isValid() then
                phys:setMass(10)  -- Light weight
                phys:wake()
            end
        end
    end
    
    -- Monitor canary prop and detect when it gets frozen (= server lag)
    local function checkCanaryProp()
        if not isValid(canaryProp) then
            createCanaryProp()
            return
        end
        
        local isFrozen = canaryProp:isFrozen()
        
        -- Detect freeze transition (was not frozen, now is frozen = LAG!)
        if isFrozen and not canaryWasFrozen then
            -- LAG DETECTED!
            local now = timer.curtime()
            
            -- Check cooldown
            if (now - lastLagReport) >= SFMA.PhysicsLag.scanCooldown then
                lastLagReport = now
                scanForPhysicsOffender()
            end
        end
        
        canaryWasFrozen = isFrozen
        
        -- If canary is frozen, unfreeze it to continue monitoring
        if isFrozen then
            canaryProp:setFrozen(false)
            
            -- Give it a little upward velocity to keep it moving
            local phys = canaryProp:getPhysicsObject()
            if phys and phys:isValid() then
                phys:setVelocity(Vector(0, 0, 100))
            end
        end
        
        -- If canary fell too far or got stuck, recreate it
        local chipPos = chip():getPos()
        local canaryPos = canaryProp:getPos()
        if canaryPos.z < chipPos.z - 200 or canaryPos.z > chipPos.z + 300 then
            createCanaryProp()
        end
    end
    
    -- Track entity creation time
    hook.add("OnEntityCreated", "SFMA_trackSpawnTime", function(ent)
        if isValid(ent) then
            entitySpawnTimes[ent] = timer.curtime()
        end
    end)
    
    -- Track collisions via damage
    hook.add("EntityTakeDamage", "SFMA_trackCollisions", function(target, attacker)
        if not SFMA.PhysicsLag.enabled then return end
        
        -- Count physics-based damage (collisions)
        local isPhysicsDamage = false
        if isValid(attacker) then
            local cls = attacker:getClass()
            if cls == "prop_physics" or cls == "worldspawn" then
                isPhysicsDamage = true
            end
        end
        
        if isPhysicsDamage and isValid(target) then
            local owner = target:getOwner()
            if isValid(owner) and owner:isPlayer() then
                collisionCounts[owner] = (collisionCounts[owner] or 0) + 1
            end
        end
    end)
    
    -- Scan all entities and find physics lag offender (OPTIMIZED - incremental scanning)
    scanForPhysicsOffender = function()
        local now = timer.curtime()
        local cfg = SFMA.PhysicsLag
        
        -- Get all entities
        local allEnts = find.all()
        local totalEnts = #allEnts
        local currentIndex = 1
        local batchSize = 50  -- Process 50 entities per tick to avoid CPU limit
        
        -- Shared state (accessible by both scanBatch and generateReport)
        local playerScores = {} -- [player] = { score = 0, details = {} }
        local playerEntityCounts = {} -- [player] = { total = 0, props = 0 }
        local chipUsage = {} -- Track all chips: { {owner, name, usage, type}, ... }
        
        -- Forward declare
        local generateReport
        
        -- Generate and display report
        generateReport = function()
            -- Calculate total scores
            local w = cfg.weights
            for player, data in pairs(playerScores) do
                data.score = 
                    (data.constraints * w.constraints) +
                    (data.velocitySpikes * w.velocitySpike) +
                    (data.heavyProps * w.heavyProp) +
                    (data.recentSpawns * w.recentSpawn) +
                    (data.whiteProps * w.whiteProp) +
                    (data.collisions * w.collisions)
            end
            
            -- Sort by score and get top 3
            local sortedPlayers = {}
            for player, data in pairs(playerScores) do
                if data.score > 0 then
                    table.insert(sortedPlayers, {player = player, data = data})
                end
            end
            
            table.sort(sortedPlayers, function(a, b) return a.data.score > b.data.score end)
            
            -- Report top 3 offenders
            SFMA.chatLog(Color(255,100,100), "⚠ PHYSICS LAG DETECTED ⚠", Color(255,255,255), " - Top suspects:")
            
            if #sortedPlayers > 0 then
                for i = 1, math.min(3, #sortedPlayers) do
                    local entry = sortedPlayers[i]
                    local ply = entry.player
                    local data = entry.data
                    
                    -- Build details string
                    local details = {}
                    if data.constraints > 0 then table.insert(details, data.constraints .. " constraints") end
                    if data.heavyProps > 0 then table.insert(details, data.heavyProps .. " heavy props") end
                    if data.recentSpawns > 0 then table.insert(details, data.recentSpawns .. " recent spawns") end
                    if data.whiteProps > 0 then table.insert(details, data.whiteProps .. " white props") end
                    if data.velocitySpikes > 0 then table.insert(details, data.velocitySpikes .. " velocity spikes") end
                    if data.collisions > 0 then table.insert(details, data.collisions .. " collisions") end
                    
                    local detailStr = table.concat(details, ", ")
                    
                    SFMA.chatLog(
                        Color(255,200,100), "#" .. i, Color(255,255,255), " - ",
                        getColor(ply), getName(ply), Color(255,255,255),
                        " (score: ", Color(255,100,100), math.floor(data.score), Color(255,255,255), ") - ",
                        Color(200,200,200), detailStr
                    )
                    
                    -- Log first offender for spectate
                    if i == 1 then
                        SFMA.logOffender(ply, true)
                    end
                end
            else
                SFMA.chatLog(Color(200,200,200), "No clear offender found based on scoring metrics.")
            end
            
            -- Entity count statistics
            local sortedEntityCounts = {}
            for player, counts in pairs(playerEntityCounts) do
                table.insert(sortedEntityCounts, {player = player, counts = counts})
            end
            table.sort(sortedEntityCounts, function(a, b) return a.counts.total > b.counts.total end)
            
            SFMA.chatLog(Color(100,200,255), "Entity Statistics", Color(255,255,255), " - Top 3 by entity count:")
            
            if #sortedEntityCounts > 0 then
                for i = 1, math.min(3, #sortedEntityCounts) do
                    local entry = sortedEntityCounts[i]
                    local ply = entry.player
                    local counts = entry.counts
                    
                    SFMA.chatLog(
                        Color(255,200,100), "#" .. i, Color(255,255,255), " - ",
                        getColor(ply), getName(ply), Color(255,255,255), ": ",
                        Color(150,255,150), counts.total, Color(255,255,255), " entities (",
                        Color(255,200,150), counts.props, Color(255,255,255), " props, ",
                        Color(200,200,255), (counts.total - counts.props), Color(255,255,255), " other)"
                    )
                end
            else
                SFMA.chatLog(Color(200,200,200), "No entity data collected.")
            end
            
            -- Chip usage statistics
            table.sort(chipUsage, function(a, b) return a.usage > b.usage end)
            
            SFMA.chatLog(Color(255,150,100), "Chip Statistics", Color(255,255,255), " - Top 3 by CPU usage:")
            
            if #chipUsage > 0 then
                for i = 1, math.min(3, #chipUsage) do
                    local chip = chipUsage[i]
                    local chipColor = chip.chipType == "SF" and Color(0,255,255) or Color(255,0,0)
                    
                    SFMA.chatLog(
                        Color(255,200,100), "#" .. i, Color(255,255,255), " - ",
                        chipColor, "[" .. chip.chipType .. "]", Color(255,255,255), " ",
                        Color(255,255,200), chip.name, Color(255,255,255), " by ",
                        getColor(chip.owner), chip.owner:getName(), Color(255,255,255), " - ",
                        Color(255,100,100), chip.usage .. "us"
                    )
                end
            else
                SFMA.chatLog(Color(200,200,200), "No chips found.")
            end
            
            -- Reset collision counts
            collisionCounts = {}
        end
        
        -- Incremental scanning function
        local function scanBatch()
            local endIndex = math.min(currentIndex + batchSize - 1, totalEnts)
            
            for i = currentIndex, endIndex do
                local ent = allEnts[i]
                
                if not isValid(ent) then goto skip_ent end
                
                local owner = ent:getOwner()
                if not isValid(owner) or not owner:isPlayer() then goto skip_ent end
                
                -- Get entity class
                local entClass = ""
                pcall(function() entClass = ent:getClass() end)
                
                -- Track chip usage (SF and E2) - count regardless of parenting
                if entClass == "starfall_processor" or entClass == "gmod_wire_expression2" then
                    local chipName = ""
                    local quota = 0
                    pcall(function() chipName = ent:getChipName() end)
                    pcall(function() quota = ent:getQuotaAverage() end)
                    quota = math.floor((quota or 0) * 1000000) -- Convert to microseconds
                    
                    if chipName == "" then chipName = "<no name>" end
                    
                    table.insert(chipUsage, {
                        owner = owner,
                        name = chipName,
                        usage = quota,
                        chipType = entClass == "starfall_processor" and "SF" or "E2"
                    })
                end
                
                -- Skip parented entities for scoring (except chips)
                local isParented = false
                pcall(function() isParented = isValid(ent:getParent()) end)
                if isParented and entClass ~= "starfall_processor" and entClass ~= "gmod_wire_expression2" then
                    goto skip_ent
                end
                
                -- Track entity counts
                if not playerEntityCounts[owner] then
                    playerEntityCounts[owner] = { total = 0, props = 0 }
                end
                playerEntityCounts[owner].total = playerEntityCounts[owner].total + 1
                
                if entClass == "prop_physics" then
                    playerEntityCounts[owner].props = playerEntityCounts[owner].props + 1
                end
                
                -- Initialize player score
                if not playerScores[owner] then
                    playerScores[owner] = {
                        score = 0,
                        constraints = 0,
                        heavyProps = 0,
                        recentSpawns = 0,
                        whiteProps = 0,
                        velocitySpikes = 0,
                        collisions = collisionCounts[owner] or 0
                    }
                end
                
                local pdata = playerScores[owner]
                
                -- Check if entity is physics-based
                local phys = nil
                pcall(function() phys = ent:getPhysicsObject() end)
                
                if phys and phys:isValid() then
                    -- 1. Count constraints (skip if already counted for this player)
                    if pdata.constraints < 1000 then  -- Cap to avoid excessive checks
                        local constrained = {}
                        pcall(function() constrained = ent:getAllConstrained() end)
                        pdata.constraints = pdata.constraints + math.min(#constrained, 100)  -- Cap individual count
                    end
                    
                    -- 2. Check for heavy props (>500kg)
                    local mass = 0
                    pcall(function() mass = phys:getMass() end)
                    if mass > cfg.heavyPropMass then
                        pdata.heavyProps = pdata.heavyProps + 1
                    end
                    
                    -- 3. Check for recent spawns (last 5 seconds)
                    local spawnTime = entitySpawnTimes[ent]
                    if spawnTime and (now - spawnTime) < cfg.recentSpawnWindow then
                        pdata.recentSpawns = pdata.recentSpawns + 1
                    end
                    
                    -- 4. Check for white-colored props (lag indicator)
                    local col = Color(0,0,0)
                    pcall(function() col = ent:getColor() end)
                    if col.r == 255 and col.g == 255 and col.b == 255 and col.a == 100 then
                        pdata.whiteProps = pdata.whiteProps + 1
                    end
                    
                    -- 5. Check velocity spikes
                    local vel = Vector(0,0,0)
                    pcall(function() vel = phys:getVelocity() end)
                    local speed = vel:getLength()
                    
                    local lastSpeed = lastVelocities[ent] or 0
                    local delta = math.abs(speed - lastSpeed)
                    
                    if delta > cfg.velocitySpikeThreshold then
                        pdata.velocitySpikes = pdata.velocitySpikes + 1
                    end
                    
                    lastVelocities[ent] = speed
                end
                
                ::skip_ent::
            end
            
            currentIndex = endIndex + 1
            
            -- If we've processed all entities, generate report
            if currentIndex > totalEnts then
                timer.remove("SFMA_scanBatch")
                generateReport()
            end
        end
        
        -- Start incremental scanning (process batches every 0.05 seconds)
        timer.create("SFMA_scanBatch", 0.05, 0, scanBatch)
    end
    
    -- Start canary prop monitoring (lightweight check every 0.15s)
    createCanaryProp()
    timer.create("SFMA_canaryMonitor", SFMA.PhysicsLag.checkRate, 0, function()
        if SFMA.PhysicsLag.enabled then
            checkCanaryProp()
        end
    end)
    
    -- Cleanup canary on chip removal
    hook.add("Removed", "SFMA_cleanupCanary", function()
        if isValid(canaryProp) then
            canaryProp:remove()
        end
    end)
    
    -- ============= END PHYSICS LAG DETECTION =============

    -- AFK Watch hook (server-side authoritative KeyPress)
    hook.add("KeyPress", "SFMA_AFKWatch", function( plr, key )
        if not SFMA.AFKWatch[plr] then return end
        SFMA.AFKWatch[plr] = nil
        SFMA.chatLog(getColor(plr), getName(plr), Color(255,255,255), " is back!")
        SFMA.logOffender( plr, true )
    end)

    -- Server receives client notifications about suspicious keys
    -- len, plr are provided by net.receive in SF
    net.receive("SFMA_keypress", function(len, plr)
        if not isValid(plr) then return end
        if not plr:isPlayer() then return end

        local ok, key = pcall(function() return net.readInt(16) end)
        if not ok or not key then return end

        local keyName = suspiciousKeys[key]
        if not keyName then return end

        -- per-player cooldown
        local now = timer.curtime()
        if lastKeyTime[plr] and (now - lastKeyTime[plr]) < KEY_PRESS_COOLDOWN then
            return
        end

    local trackedProppushProps = {}

    local function checkHolding( ent )
        if not isValid(ent) or not ent:isPlayerHolding() then return table.removeByValue(trackedProppushProps, ent) end
        if ent:getCollisionGroup() == COLLISION_GROUP.WORLD then return end
        if proppushAlertCooldown[ent.LastHolder] then return end

        local aabb_min, aabb_max = ent:worldSpaceAABB()
        local lookahead = ent:getVelocity() * 0.1

        local pushing = find.inBox( aabb_min + lookahead, aabb_max + lookahead, function( e )
            if not isValid( e ) then return end
            if e == ent or e == ent.LastHolder then return end

            local owner = e:getOwner()
            if owner == ent.LastHolder then return end

            if e:isPlayer() then
                if e:isNoclipped() then return end
                if isValid(e:getVehicle()) then return end
            else
                if not isValid( e:getPhysicsObject() ) then return end
                if not isValid( owner ) then return end
                if e:isFrozen() then return end
                if PP_BLACKLIST[e:getClass()] then return end
            end

            return true
        end)

        if #pushing > 0 then
            SFMA.chatLog("Detected ", getColor(ent.LastHolder), getName(ent.LastHolder), Color(255,255,255), " prop pushing!")
            SFMA.logOffender( ent.LastHolder , true )
            proppushAlertCooldown[ent.LastHolder] = true

            printConsole("wp: ", ent.LastHolder:getActiveWeapon())
            printConsole("src ent: ", ent)
            printConsole("src held?: ", ent:isPlayerHolding())
            printConsole("pped ent: ")
            for i,v in pairs(pushing) do
                printConsole(tostring(i) .. ". " .. tostring(v) .. " - ", v:getOwner(), " - " .. (v:getPos():getDistance(ent:getPos()) .. "u"))
            end

            timer.simple(30, function()
                proppushAlertCooldown[ent.LastHolder] = nil
            end)
        end
    end

    timer.create( "SFMA_proppushDetection", 0.25, 0, function()

        for _, ent in pairs(trackedProppushProps) do checkHolding(ent) end
    end)

        --ensureFont()
        local now = timer.curtime()
        local lh = SFMA.HUD.lineHeight

        -- get screen resolution (works in your other chip)
        local screenW, screenH = render.getGameResolution()

        -- remove expired messages (iterate from back to front)
        for i = #Notifications, 1, -1 do
            local n = Notifications[i]
            if not n then
                table.remove(Notifications, i)
            else
                local elapsed = now - n.t
                if elapsed >= n.dur then
                    table.remove(Notifications, i)
                end
            end
        end

        local num = math.min(#Notifications, SFMA.HUD.maxMessages)
        if num == 0 then return end

        -- compute vertical centered start Y (middle of left side)
        local totalHeight = num * lh
        local startY = math.floor(screenH * 0.5 - totalHeight * 0.5)

        local x = SFMA.HUD.x
        local bgw = SFMA.HUD.bgWidth

        -- draw from newest (index 1) downward
        for i = 1, num do
            local n = Notifications[i]
            if not n then break end
            local elapsed = now - n.t
            local alpha = math.floor(255 * (1 - (elapsed / n.dur)))
            if alpha < 0 then alpha = 0 end
            local col = Color(n.color.r, n.color.g, n.color.b, alpha)
            local y = startY + (i-1) * lh

            render.setFont(hudFont)
            -- background rect for readability (slightly translucent black)
            render.setColor(Color(0,0,0, math.floor(alpha * 0.6)))
            render.drawRect(x - 6, y - 2, bgw, SFMA.HUD.bgHeight)
            -- text (left aligned)

        end
